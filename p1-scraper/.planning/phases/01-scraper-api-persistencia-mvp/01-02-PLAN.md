Use `uuid_generate_v4()` is NOT guaranteed; use `gen_random_uuid()` for `id` primary keys (pgcrypto/built-in in Supabase).

```sql
-- p1_events
create table if not exists public.p1_events (
  id uuid primary key default gen_random_uuid(),
  event_uuid uuid not null unique,
  source_url text,
  slug text,
  name text,
  series text default 'formula-1',
  status text,
  venue_name text,
  city text,
  country_code text,
  lat double precision,
  lng double precision,
  time_zone text,
  date_time timestamptz,
  date_time_end timestamptz,
  main_image_url text,
  main_image_storage_path text,
  marketing_label text,
  description text,
  price_ticket_only numeric,
  price_ticket_hotel numeric,
  price_compare numeric,
  currency text default 'EUR',
  tc_ticket_id text,
  active boolean default true,
  first_seen_at timestamptz default now(),
  last_seen_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- p1_tickets
create table if not exists public.p1_tickets (
  id uuid primary key default gen_random_uuid(),
  event_id uuid not null references public.p1_events(id) on delete cascade,
  category_id uuid not null,
  name text,
  description text,
  seatplan_image_url text,
  seatplan_image_storage_path text,
  price numeric,
  currency text default 'EUR',
  features jsonb,
  delivery_type text,
  availability text default 'available',
  last_seen_at timestamptz default now(),
  updated_at timestamptz default now(),
  unique (event_id, category_id)
);

-- p1_price_history
create table if not exists public.p1_price_history (
  id uuid primary key default gen_random_uuid(),
  ticket_id uuid not null references public.p1_tickets(id) on delete cascade,
  price numeric,
  currency text default 'EUR',
  recorded_at timestamptz default now()
);

create index if not exists idx_p1_tickets_event_id on public.p1_tickets(event_id);
create index if not exists idx_p1_price_history_ticket_id on public.p1_price_history(ticket_id);
```
